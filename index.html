#include <thwart/scanner.hpp>

uintptr_t ResolveLocalPlayer(HANDLE process_handle) {
    const char* pattern = "\x48\x8B\x0D\x00\x00\x00\x00\x48\x85\xC9";
    const char* mask = "xxx????xxx";

    uintptr_t match = thwart::scanner::ScanPattern(process_handle, pattern, mask);
    if (!match) {
        thwart::LogWarning("Pattern match failed, retrying on next tick.");
        return 0;
    }

    return thwart::memory::ReadRelativePointer(match + 3);
}
